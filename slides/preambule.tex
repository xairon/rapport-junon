% --- Captures de la plateforme ---------------------------------------------
% Les originaux de figures/ sont des fenetres de 3840 px : projetees en entier,
% leur texte d'interface mesure environ 1 mm et ne se lit pas. Les vues de
% figures/vues/ cadrent la zone qui porte le propos (voir figures/recadrer.sh).
% Un filet clair detache la capture, qui est sombre, du fond blanc.
%
% Usage : \vue[largeur relative]{nom-de-vue}
% Le filet compte dans la largeur du resultat : sans le \dimexpr, une vue en
% pleine largeur deborde de 2\fboxrule.
\newcommand{\vue}[2][1.0]{%
  {\setlength{\fboxsep}{0pt}\setlength{\fboxrule}{0.4pt}%
   \fcolorbox{black!30}{white}{\includegraphics[width=\dimexpr#1\linewidth-2\fboxrule\relax]%
     {../figures/vues/#2.png}}}}

% Variante bornee en largeur ET en hauteur : sur une diapositive, c'est la
% hauteur qui manque, et une vue large deborde sous le pied de page sans que
% LaTeX ne signale autre chose qu'une \vbox trop haute.
% Usage : \vueb[largeur relative]{hauteur en fraction de \paperheight}{nom}
\newcommand{\vueb}[3][1.0]{%
  {\setlength{\fboxsep}{0pt}\setlength{\fboxrule}{0.4pt}%
   \fcolorbox{black!30}{white}{\includegraphics[width=\dimexpr#1\linewidth-2\fboxrule\relax,
     height=#2\paperheight,keepaspectratio]{../figures/vues/#3.png}}}}

% Diapositive dont la capture est le propos : image au plus grand, puis une
% seule ligne qui dit ce qu'il faut y lire. Rien d'autre, jamais.
% Usage : \cadre{titre}{nom-de-vue}{cle de lecture}
\newcommand{\cadre}[3]{%
\begin{frame}{#1}
  \vspace{-0.15cm}
  \begin{center}\vueb[0.98]{0.575}{#2}\end{center}
  \vspace{0.15cm}
  {\scriptsize\color{grisardoise} #3\par}
\end{frame}}

% Variante pour une vue peu allongee : sur une diapositive 16:9, une image de
% format 3:2 bornee en hauteur laisse un tiers de la largeur vide. La cle de
% lecture occupe alors la colonne de droite au lieu du dessous.
% Usage : \cadrecote{titre}{nom-de-vue}{cle de lecture}
\newcommand{\cadrecote}[3]{%
\begin{frame}{#1}
  \vspace{-0.15cm}
  \begin{columns}[T]
  \begin{column}{0.60\textwidth}
    \vueb{0.62}{#2}
  \end{column}
  \begin{column}{0.37\textwidth}
    \vspace{0.2cm}
    {\footnotesize\color{grisardoise} #3\par}
  \end{column}
  \end{columns}
\end{frame}}

% Deux vues cote a cote, une legende chacune. Les images sont centrees dans une
% bande de hauteur fixe, faute de quoi deux formats differents desalignent les
% legendes.
% Usage : \cadredouble{titre}{vue1}{legende1}{vue2}{legende2}
\newcommand{\cadredouble}[5]{%
\begin{frame}{#1}
  \vspace{-0.1cm}
  \begin{columns}[T]
  \begin{column}{0.485\textwidth}
    \parbox[t][0.40\paperheight][c]{\linewidth}{\centering\vueb{0.40}{#2}}
    \vspace{0.1cm}
    {\scriptsize\color{grisardoise} #3\par}
  \end{column}
  \begin{column}{0.485\textwidth}
    \parbox[t][0.40\paperheight][c]{\linewidth}{\centering\vueb{0.40}{#4}}
    \vspace{0.1cm}
    {\scriptsize\color{grisardoise} #5\par}
  \end{column}
  \end{columns}
\end{frame}}
