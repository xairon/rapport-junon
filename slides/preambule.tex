% --- Emplacement de capture -------------------------------------------------
% Affiche la capture si le fichier existe, sinon un cadre qui dit precisement
% ce qu'il faut photographier. Les diapositives compilent donc avant que les
% captures soient prises, et se completent sans retouche quand elles arrivent.
% Usage : \capture[largeur]{nom-du-fichier}{ce qu'il faut montrer}
\newcommand{\capture}[3][0.86]{%
  \begin{center}
  \IfFileExists{captures/#2}%
    {\includegraphics[width=#1\linewidth,keepaspectratio]{captures/#2}}%
    {\begin{tikzpicture}
       \node[draw=grisardoise, dashed, thick, fill=grisclair, rounded corners=2pt,
             minimum width=#1\linewidth, minimum height=0.40\paperheight,
             align=center, text width=0.8\linewidth, font=\small\color{grisardoise}]
            {\textbf{Capture à insérer} : \code{#2}\\[0.4em] #3};
     \end{tikzpicture}}%
  \end{center}}
